// K-Talk 채팅 자동 음성 읽기 let ktalkVoiceOn = true; function readChat(text, important = false) { if (!ktalkVoiceOn || !text) return; // 중요한 알림은 기존 읽기를 멈추고 먼저 읽기 if (important) { speechSynthesis.cancel(); } const msg = new SpeechSynthesisUtterance(text); msg.lang = 'ko-KR'; msg.rate = 1.0; msg.pitch = 1.0; msg.volume = 1.0; speechSynthesis.speak(msg); } // 일반 채팅 function newChat(nickname, text) { readChat(nickname + "님, " + text); } // 중요 공지 function importantNotice(text) { readChat("중요 안내입니다. " + text, true); } // 경고 function warningNotice(text) { readChat("경고 안내입니다. " + text, true); } // 선물 알림 function giftNotice(nickname, gift) { readChat(nickname + "님이 " + gift + " 선물을 보냈습니다.", true); }
